#!/usr/bin/env bash
set -euo pipefail
exec 9>/run/agent-rush-requalify-box.lock
flock -n 9 || { echo "box requalification already running" >&2; exit 0; }
set -a; . /etc/agent-rush/worker.env; set +a
out=/var/lib/agent-rush/box-qualification.next.json
rm -f "$out"
if runuser -u agent-rush -- /opt/agent-rush/agent-rush qualify-box --output "$out" --private-probe "10.42.0.3:22" --listener-port 8090 --listener-port 8443 --listener-port 22; then
  mv -f "$out" /var/lib/agent-rush/box-qualification.json
else
  echo "box qualification failed; boxes stay refused once the current report expires" >&2
  exit 1
fi
---
-N DOCKER-USER
-A DOCKER-USER -s 172.30.0.0/24 -d 0.0.0.0/8 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited

-A DOCKER-USER -s 172.30.0.0/24 -d 127.0.0.0/8 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d 169.254.0.0/16 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d 172.16.0.0/12 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d 192.168.0.0/16 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d 224.0.0.0/4 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d 240.0.0.0/4 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d WORKER-PUBLIC-IP-REDACTED/32 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d 10.42.0.4/32 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d 172.17.0.1/32 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -d 172.30.0.1/32 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
-A DOCKER-USER -s 172.30.0.0/24 -p tcp -m multiport --dports 25,465,587 -m comment --comment ar-box-egress -j REJECT --reject-with tcp-reset
-A DOCKER-USER -s 172.30.0.0/24 -m conntrack --ctstate NEW -m hashlimit --hashlimit-above 50/sec --hashlimit-burst 100 --hashlimit-mode srcip --hashlimit-name ar-box -m comment --comment ar-box-egress -j DROP
-A DOCKER-USER -s 172.30.0.0/24 -m comment --comment ar-box-egress -j RETURN
-A INPUT -s 172.30.0.0/24 -m comment --comment ar-box-egress -j REJECT --reject-with icmp-admin-prohibited
---
server {
    listen 10.42.0.4:8443 ssl;
    server_name worker.agent-rush.internal;
    ssl_certificate /etc/agent-rush/tls/worker.crt;
    ssl_certificate_key /etc/agent-rush/tls/worker.key;
    ssl_client_certificate /etc/agent-rush/tls/ca.crt;
    ssl_verify_client on;
    ssl_verify_depth 2;
    ssl_protocols TLSv1.2 TLSv1.3;
    client_max_body_size 64k;
    # Box calls carry files up to 25 MiB and commands that run up to 600 s.
    location /v1/boxes {
        client_max_body_size 26m;
        proxy_pass http://127.0.0.1:8090;
        proxy_set_header Host $host;
        proxy_set_header Authorization $http_authorization;
        proxy_connect_timeout 3s;
        proxy_read_timeout 660s;
        proxy_send_timeout 120s;
    }
    location / {
        proxy_pass http://127.0.0.1:8090;
        proxy_set_header Host $host;
        proxy_set_header Authorization $http_authorization;
        proxy_connect_timeout 3s;
        proxy_read_timeout 15s;
    }
}
